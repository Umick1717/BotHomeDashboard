const NUICE_SPREADSHEET_ID =
  "1_G5dDFWUcAb_dANQ8GVF5VL6v3QDTOxhJD8Fc0nUPos";

const NUICE_SHEET_ID = 34336380;

const NUICE_LAYOUT = Object.freeze({
  income: {
    startRow: 33,
    startColumn: 4,
    dateColumn: 4,
    amountColumn: 6
  },

  expense: {
    startRow: 118,
    startColumn: 10,
    dateColumn: 10,
    amountColumn: 12
  }
});

function doGet(e) {
  try {
    const ss = SpreadsheetApp.openById(NUICE_SPREADSHEET_ID);
    const sheet = ss.getSheetById(NUICE_SHEET_ID);

    return jsonOutput_({
      ok: true,
      name: "NuIce Expense API V15",
      spreadsheetId: ss.getId(),
      spreadsheetName: ss.getName(),
      sheetId: NUICE_SHEET_ID,
      sheetFound: Boolean(sheet),
      sheetName: sheet ? sheet.getName() : ""
    });
  } catch (error) {
    return jsonOutput_({
      ok: false,
      name: "NuIce Expense API V15",
      error: String(error && error.message ? error.message : error)
    });
  }
}

function doPost(e) {
  try {
    const data = parseRequest_(e);
    const result = saveNuIceTransaction_(data);

    return jsonOutput_({
      ok: true,
      status: "success",
      ...result
    });
  } catch (error) {
    console.error(error);

    return jsonOutput_({
      ok: false,
      status: "error",
      error: String(error && error.message ? error.message : error)
    });
  }
}

function parseRequest_(e) {
  if (!e) {
    throw new Error("ไม่พบ Event จาก Web App");
  }

  const p = e.parameter || {};

  if (p.type || p.date || p.item || p.amount || p.group) {
    return {
      tracker: String(p.tracker || "nuice"),
      type: String(p.type || ""),
      date: String(p.date || ""),
      item: String(p.item || ""),
      amount: Number(String(p.amount || "0").replace(/,/g, "")),
      group: String(p.group || "")
    };
  }

  if (e.postData && e.postData.contents) {
    const raw = String(e.postData.contents || "").trim();

    if (raw) {
      return JSON.parse(raw);
    }
  }

  throw new Error("ไม่พบข้อมูล POST");
}

function saveNuIceTransaction_(data) {
  validatePayload_(data);

  const lock = LockService.getScriptLock();
  lock.waitLock(15000);

  try {
    const ss = SpreadsheetApp.openById(NUICE_SPREADSHEET_ID);
    const sheet = ss.getSheetById(NUICE_SHEET_ID);

    if (!sheet) {
      throw new Error("ไม่พบ Sheet ID " + NUICE_SHEET_ID);
    }

    const layout = NUICE_LAYOUT[data.type];

    if (!layout) {
      throw new Error("ประเภทข้อมูลไม่ถูกต้อง");
    }

    const row = getNextEmptyRow_(
      sheet,
      layout.startColumn,
      layout.startRow
    );

    const date = parseLocalISODate_(data.date);

    sheet
      .getRange(row, layout.startColumn, 1, 4)
      .setValues([[
        date,
        String(data.item || "").trim(),
        Number(data.amount),
        String(data.group || "").trim()
      ]]);

    sheet
      .getRange(row, layout.dateColumn)
      .setNumberFormat("dd/MM/yyyy");

    sheet
      .getRange(row, layout.amountColumn)
      .setNumberFormat("#,##0.00");

    SpreadsheetApp.flush();

    return {
      type: data.type,
      row: row,
      sheetName: sheet.getName(),
      savedAt: new Date().toISOString()
    };
  } finally {
    lock.releaseLock();
  }
}

function validatePayload_(data) {
  if (!data || typeof data !== "object") {
    throw new Error("รูปแบบข้อมูลไม่ถูกต้อง");
  }

  if (data.type !== "income" && data.type !== "expense") {
    throw new Error("type ต้องเป็น income หรือ expense");
  }

  if (
    data.tracker &&
    String(data.tracker).toLowerCase() !== "nuice"
  ) {
    throw new Error("Tracker ไม่ถูกต้อง");
  }

  if (!/^\d{4}-\d{2}-\d{2}$/.test(String(data.date || ""))) {
    throw new Error("วันที่ไม่ถูกต้อง");
  }

  if (!String(data.item || "").trim()) {
    throw new Error("กรุณากรอกชื่อรายการ");
  }

  const amount = Number(data.amount);

  if (!isFinite(amount) || amount <= 0) {
    throw new Error("จำนวนเงินไม่ถูกต้อง");
  }

  if (!String(data.group || "").trim()) {
    throw new Error("กรุณาเลือกกลุ่ม");
  }
}

function parseLocalISODate_(isoDate) {
  const parts = String(isoDate).split("-").map(Number);

  return new Date(
    parts[0],
    parts[1] - 1,
    parts[2],
    12,
    0,
    0
  );
}

function getNextEmptyRow_(sheet, column, startRow) {
  const lastRow = Math.max(sheet.getLastRow(), startRow);
  const numberOfRows = lastRow - startRow + 1;

  const values = sheet
    .getRange(startRow, column, numberOfRows, 1)
    .getDisplayValues();

  for (let index = 0; index < values.length; index++) {
    if (String(values[index][0] || "").trim() === "") {
      return startRow + index;
    }
  }

  return lastRow + 1;
}

function jsonOutput_(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
