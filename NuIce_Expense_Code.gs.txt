/**
 * My Expense Tracker 2026 Nu'Ice
 * Google Apps Script Backend V14
 *
 * Target Spreadsheet:
 * 1_G5dDFWUcAb_dANQ8GVF5VL6v3QDTOxhJD8Fc0nUPos
 * Target Sheet GID / Sheet ID: 34336380
 *
 * Layout follows the existing Mick tracker:
 * Income  = D:G starting row 33
 * Expense = J:M starting row 118
 */

const NUICE_SPREADSHEET_ID =
  "1_G5dDFWUcAb_dANQ8GVF5VL6v3QDTOxhJD8Fc0nUPos";

const NUICE_SHEET_ID = 34336380;

const NUICE_LAYOUT = Object.freeze({
  income: {
    startRow: 33,
    startColumn: 4,
    dateColumn: 4,
    amountColumn: 6
  },
  expense: {
    startRow: 118,
    startColumn: 10,
    dateColumn: 10,
    amountColumn: 12
  }
});

function doGet(e) {
  const action = e && e.parameter
    ? String(e.parameter.action || "")
    : "";

  if (action === "health") {
    return jsonOutput_({
      ok: true,
      name: "NuIce Expense API V14",
      spreadsheetId: NUICE_SPREADSHEET_ID,
      sheetId: NUICE_SHEET_ID
    });
  }

  return ContentService
    .createTextOutput("NuIce Expense API V14 is running")
    .setMimeType(ContentService.MimeType.TEXT);
}

function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      throw new Error("ไม่พบข้อมูลที่ส่งมา");
    }

    const data = JSON.parse(e.postData.contents);
    const result = saveNuIceTransaction_(data);

    return jsonOutput_({
      ok: true,
      ...result
    });
  } catch (error) {
    console.error(error);

    return jsonOutput_({
      ok: false,
      error: String(error && error.message ? error.message : error)
    });
  }
}

function saveNuIceTransaction_(data) {
  validatePayload_(data);

  const ss = SpreadsheetApp.openById(NUICE_SPREADSHEET_ID);
  const sheet = ss.getSheetById(NUICE_SHEET_ID);

  if (!sheet) {
    throw new Error("ไม่พบ Sheet ID " + NUICE_SHEET_ID);
  }

  const layout = NUICE_LAYOUT[data.type];

  if (!layout) {
    throw new Error("ประเภทข้อมูลไม่ถูกต้อง");
  }

  const row = getNextEmptyRow_(
    sheet,
    layout.startColumn,
    layout.startRow
  );

  const date = parseLocalISODate_(data.date);

  sheet
    .getRange(row, layout.startColumn, 1, 4)
    .setValues([[
      date,
      String(data.item || "").trim(),
      Number(data.amount),
      String(data.group || "").trim()
    ]]);

  sheet.getRange(row, layout.dateColumn).setNumberFormat("dd/MM/yyyy");
  sheet.getRange(row, layout.amountColumn).setNumberFormat("#,##0.00");

  SpreadsheetApp.flush();

  return {
    type: data.type,
    row,
    savedAt: new Date().toISOString()
  };
}

function validatePayload_(data) {
  if (!data || typeof data !== "object") {
    throw new Error("รูปแบบข้อมูลไม่ถูกต้อง");
  }

  if (data.type !== "income" && data.type !== "expense") {
    throw new Error("type ต้องเป็น income หรือ expense");
  }

  if (
    data.tracker &&
    String(data.tracker).toLowerCase() !== "nuice"
  ) {
    throw new Error("Tracker ไม่ถูกต้อง");
  }

  if (!/^\d{4}-\d{2}-\d{2}$/.test(String(data.date || ""))) {
    throw new Error("วันที่ไม่ถูกต้อง");
  }

  if (!String(data.item || "").trim()) {
    throw new Error("กรุณากรอกชื่อรายการ");
  }

  const amount = Number(data.amount);

  if (!Number.isFinite(amount) || amount <= 0) {
    throw new Error("จำนวนเงินไม่ถูกต้อง");
  }

  if (!String(data.group || "").trim()) {
    throw new Error("กรุณาเลือกกลุ่ม");
  }
}

function parseLocalISODate_(isoDate) {
  const parts = String(isoDate).split("-").map(Number);

  return new Date(
    parts[0],
    parts[1] - 1,
    parts[2],
    12,
    0,
    0
  );
}

function getNextEmptyRow_(sheet, column, startRow) {
  const lastRow = Math.max(sheet.getLastRow(), startRow);
  const rowCount = lastRow - startRow + 1;

  const values = sheet
    .getRange(startRow, column, rowCount, 1)
    .getDisplayValues();

  for (let index = 0; index < values.length; index += 1) {
    if (String(values[index][0] || "").trim() === "") {
      return startRow + index;
    }
  }

  return lastRow + 1;
}

function jsonOutput_(value) {
  return ContentService
    .createTextOutput(JSON.stringify(value))
    .setMimeType(ContentService.MimeType.JSON);
}
